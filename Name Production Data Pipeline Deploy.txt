name: Production Data Pipeline Deployment

on:
  push:
    branches:
      - main
    paths:
      - 'databricksrepo/**' 
jobs:
  deploy-to-gcs:
    name: Sync Code to Google Cloud Storage
    runs-on: ubuntu-latest
    permissions:
      contents: 'read'
      id-token: 'write'

    steps:
      - name: Checkout Code Repository
        uses: actions/checkout@v4

      - name: Authenticate to Google Cloud
        uses: google-github-actions/auth@v2
        with:
          credentials_json: ${{ secrets.GCP_SA_KEY }}

      - name: Set up Cloud SDK
        uses: google-github-actions/setup-gcloud@v2

      # Target the actual file path inside your workspace
      - name: Upload Script to GCS Staging Bucket
        run: |
          gcloud storage cp ./databricksrepo/main.py gs://your-production-code-bucket/scripts/main.py
